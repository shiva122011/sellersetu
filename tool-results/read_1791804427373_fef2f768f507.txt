    14→// Same AES-256-GCM scheme as src/lib/crypto.ts
    15→function encryptVault(payload: Record<string, string>) {
    16→  const secret = process.env.AUTH_SECRET || 'sellersetu-dev-secret-change-in-production'
    17→  const key = crypto.scryptSync(secret, 'sellersetu-credential-vault-v1', 32)
    18→  const iv = crypto.randomBytes(12)
    19→  const cipher = crypto.createCipheriv('aes-256-gcm', key, iv)
    20→  const encrypted = Buffer.concat([cipher.update(JSON.stringify(payload), 'utf8'), cipher.final()])
    21→  return {
    22→    cipherText: encrypted.toString('base64'),
    23→    iv: iv.toString('base64'),
    24→    authTag: cipher.getAuthTag().toString('base64'),
    25→  }
    26→}
